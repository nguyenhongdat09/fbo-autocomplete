const vscode = require('vscode');
const path = require('path');
const { execFile, exec } = require('child_process');
const { promisify } = require('util');
class TranslatedText {

    constructor() {
        this.fr_lan = 'vi'
        this.to_lan = 'en'
        this.one_file_path = path.join(__dirname, './one_file_translate.exe');
        this.word_path = path.join(__dirname, './word_translate.exe');
    }

    async trans_to_vi(text_to_trans) {
        //const { text } = await translate(text_to_trans, { from: 'vi', to: 'en' });
        if (text_to_trans == "")
            return ""
        try {
            const execFileAsync = promisify(execFile);
            const { stdout } = await execFileAsync(this.word_path, [text_to_trans]);
            return stdout.trim();
        }catch(error){
            return "unidefine";
        } 
    }
    
    async  translateXmlFile() {
        const editor = vscode.window.activeTextEditor;
        if (!editor) {
            vscode.window.showErrorMessage('Please open an XML file to translate.');
            return;
        }  
        // Regular Expression to find pattern: v="some text" e=""
        const  filePath = editor.document.fileName; 
        vscode.window.withProgress({
                        location: vscode.ProgressLocation.Notification,
                        title: `Translating`,
                        cancellable: false
        }, async (progress, token) => {  
            exec(`"${this.one_file_path}" "${filePath}"  `, (error, stdout, stderr) => {
                if (error) {
                    vscode.window.showErrorMessage(`Lỗi: ${error.message}`);
                    return;
                }
                if (stderr) {
                    vscode.window.showErrorMessage(`Cảnh báo: ${stderr}`);
                    return;
                } 
            });
        }); 
    }
    
}   

module.exports = TranslatedText
